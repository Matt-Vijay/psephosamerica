class BudgetAcquirer(Acquirer):
    """Persist decoded bytes after every fetch, including policies and failed attempts."""
    def __init__(self, store):
        self.ledger_path = BASE / 'http-budget.json'
        self.ledger = json.loads(self.ledger_path.read_text()) if self.ledger_path.exists() else {
            'cap': CAP, 'newly_decoded_bytes': 0, 'runs': []}
        self.prior = self.ledger['newly_decoded_bytes']
        # Reserve 1MiB against a final decoded chunk crossing Acquirer's limit.
        super().__init__(store, max_bytes=max(0, CAP-self.prior-1048576), delay=1.1)
        self.started = utc_now()
    def _pause(self, url, delay=None):
        if urlsplit(url).hostname != 'legislation.nysenate.gov':
            raise AcquisitionError('Unapproved redirect/host: ' + url)
        if shutil.disk_usage(BASE).free < 100 * 1024**3:
            raise AcquisitionError('Stopped at 100 GiB disk-free threshold')
        super()._pause(url, delay)
    def fetch(self, url, **kwargs):
        try:
            return super().fetch(url, **kwargs)
        finally:
            self.ledger['newly_decoded_bytes'] = self.prior + self.downloaded
            self.ledger['active_run'] = {'started': self.started, 'bytes': self.downloaded}
            tmp = self.ledger_path.with_suffix('.tmp')
            tmp.write_text(json.dumps(self.ledger, indent=2))
            tmp.replace(self.ledger_path)


def parse_pdf(path: Path, law_id: str, title: str, url: str):
    if path.read_bytes()[:5] != b'%PDF-':
        raise ValueError('HTTP 200 was not a PDF')
    info = subprocess.run(['pdfinfo', str(path)], capture_output=True, text=True, check=True).stdout
    expected = int(re.search(r'^Pages:\s+(\d+)', info, re.M)[1])
    text = subprocess.run(['pdftotext', '-layout', '-enc', 'UTF-8', str(path), '-'],
                          capture_output=True, text=True, check=True).stdout
    pages = text.split('\f')
    if not pages[-1].strip(): pages.pop()
    if len(pages) != expected: raise ValueError('Extracted PDF page count mismatch')
    combined = '\n'.join(pages)
    # Fail closed on a catalog-only root PDF or a server's PDF error message.
    substantive = re.findall(r'(?m)^\s*(?:§\s*\d|Section\s+\d[\dA-Za-z-]*\.|S\s+\d+\.)', combined)
    if len(substantive) < 2 or len(combined) < 1500 or len(re.findall(r'\b(?:shall|must)\b', combined, re.I)) < 3:
        raise ValueError('No demonstrated substantive statutory text: not counted')
    identity_words = re.findall(r'[a-z]{4,}', title.lower())
    first = '\n'.join(pages[:min(25,len(pages))]).lower()
    if not all(w in first for w in identity_words if w not in {'state','york','services'}):
        raise ValueError('Expected volume identity not established in opening pages: '+title)
    units=[]
    empty=[]
    for num, page in enumerate(pages,1):
        body = page.strip('\r\n')
        if not body.strip():
            empty.append(num)
            continue
        units.append(Provision(key=f'ny:{law_id}:page:{num}',
            citation=f'NY {title} — PDF page {num}', heading=f'{title} — page {num}',
            text=body, markup='<pre>'+html.escape(body)+'</pre>', url=f'{url}#page={num}',
            unit_kind='page', metadata={
                'source_law_id':law_id,'pdf_page':num,'pdf_pages':expected,
                'text_basis':'Poppler pdftotext -layout; no OCR',
                'page_is_not_a_statutory_section':True,
                'may_include_contents_or_headings':True,
                'media_status':'Original PDF authoritative for visual layout and any non-text graphics; not transcribed',
            }))
    return units, {'pdf_pages':expected,'indexed_nonempty_pages':len(units),
                   'empty_pages_not_indexed':empty,'substantive_section_starts_detected':len(substantive),
                   'pdf_info':info,'text_characters':len(combined)}


def sync_ny(store: Store, acquirer: Acquirer, limit: int | None = None, as_of: str | None = None):
    if as_of is not None:
        raise ValueError('Historical as_of not supported by this public PDF adapter; no fabricated snapshot date')
    acquirer.delay = max(acquirer.delay, 1.1)
    store.collection('ny-state-laws', ('us-ny','New York','state','us'),
        name='New York State Laws — selected complete Senate PDF volumes',
        authority='New York State Senate / Legislative Bill Drafting Commission',
        kind='statutes_and_constitution', homepage=ORIGIN,
        source_status='Official publisher source; selected volume scope, PDF page indexing',
        access=ACCESS, metadata={'scope':'Explicit selected complete volumes; not statewide exhaustive',
            'inventory_basis':'Manual selected list; statewide JSON inventory inaccessible without account',
            'clock_note':'PDF export gives no established incorporation or snapshot date; acquisition clock only',
            'documentation':DOCS})
    for code,title in VOLUMES:
        existing=store.db.execute('select status from inventories where collection_id=? and item=?',
                                 ('ny-state-laws',code)).fetchone()
        if not existing:
            store.inventory('ny-state-laws',code,f'{ORIGIN}/pdf/laws/{code}?full=true','pending')
    acquirer.fetch(DOCS, max_file_bytes=1024**2)
    acquirer.fetch(ORIGIN+'/static/docs/html/index.html', max_file_bytes=1024**2)
    results=[]
    for code,title in VOLUMES[:limit] if limit is not None else VOLUMES:
        url=f'{ORIGIN}/pdf/laws/{code}?full=true'
        previous = store.db.execute(
            'select status,error from inventories where collection_id=? and item=?',
            ('ny-state-laws',code)).fetchone()
        if previous and previous['status']=='failed' and '404' in (previous['error'] or '') and not acquirer.refresh:
            result={'law_id':code,'created':False,'skipped':'Previously recorded 404; use explicit refresh to retry'}
            results.append(result)
            print(json.dumps(result),flush=True)
            continue
        try:
            receipt=acquirer.fetch(url,max_file_bytes=min(32*1024**2, max(1,acquirer.max_bytes-acquirer.downloaded)))
            units, metadata=parse_pdf(store.object_path(receipt.sha256),code,title,url)
            version,count,created=store.ingest(collection='ny-state-laws',document=f'ny:{code}',
                title=title,url=url,acquisition=receipt.id,parser=PARSER,provisions=units,
                snapshot_basis='Unknown publisher incorporation/snapshot date; public live whole-law PDF export',
                metadata={**metadata,'source_law_id':code,'scope':'complete requested publisher volume export',
                          'effective_date_note':'Dates within legal text retained; no blanket effective date inferred',
                          'edition_date':None,'publisher_incorporation_date':None})
            store.inventory('ny-state-laws',code,url,'ingested')
            result={'law_id':code,'pages':count,'created':created,'sha256':receipt.sha256,'version':version}
        except Exception as exc:
            store.inventory('ny-state-laws',code,url,'failed',str(exc))
            result={'law_id':code,'error':str(exc)}
        results.append(result)
        print(json.dumps(result),flush=True)
        write_report(store)
        if acquirer.max_bytes-acquirer.downloaded < 2*1024**2: break
    return results


def write_report(store):
    counts={t:store.db.execute(f'select count(*) from {t}').fetchone()[0]
            for t in ['collections','documents','versions','provisions','artifacts','acquisitions']}
    receipts=[dict(r) for r in store.db.execute('select a.*,b.bytes from acquisitions a left join artifacts b on b.sha256=a.sha256 order by a.id')]
    inventory=[dict(r) for r in store.db.execute('select * from inventories order by item')]
    raw=store.db.execute('select coalesce(sum(bytes),0) from artifacts').fetchone()[0]
    ledger=json.loads((BASE/'http-budget.json').read_text())
    report={'generated_at':utc_now(),'store_path':str(store.root),'adapter_path':str(BASE/'adapter.py'),
        'baseline':'e077e981d1e7f9914e17f73407b557973ee0ef15','counts':counts,
        'unit_kind':'PDF page; counts are not statutory sections, and include contents pages within full volumes',
        'unique_raw_bytes':raw,'newly_decoded_http_bytes':ledger['newly_decoded_bytes'],
        'hard_http_cap':CAP,'budget_remaining':CAP-ledger['newly_decoded_bytes'],
        'http_accounting':'All network acquisition via Acquirer; includes successful probes/policies. Error and redirect bodies not decoded. No other HTTP tools used.',
        'coverage':inventory,'access_conditions':ACCESS,'receipts':receipts,
        'clocks':{'publisher_snapshot_date':None,'publisher_incorporation_date':None,
                  'blanket_effective_date':None,'acquisition':'per-receipt observed_at UTC',
                  'date_policy':'Unknown stays unknown; PDF generation/HTTP timestamps are not legal currency'},
        'gaps':['Not statewide exhaustive: selected volumes only; unavailable authenticated inventory prevents inventory completeness claims.',
                'No NYCRR regulations, session laws, local enactments, or NYC-specific collections.',
                'PDF text is a layout-preserving page projection, not section-level structured law.',
                'Graphics and visual tables require retained original PDF; no OCR or visual transcription.',
                'Root-only CNS PDF is retained as a probe/catalog and is not an indexed full-text version.'],
        'verification_file':str(BASE/'checks.json')}
    tmp=BASE/'report.tmp';tmp.write_text(json.dumps(report,indent=2));tmp.replace(BASE/'report.json')
