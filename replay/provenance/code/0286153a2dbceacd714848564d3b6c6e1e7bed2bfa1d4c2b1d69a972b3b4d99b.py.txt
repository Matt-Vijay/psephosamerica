from pathlib import Path
import json
import shutil
from urllib.parse import urlsplit
from psephos.acquire import Acquirer, AcquisitionError
from psephos.store import Store, utc_now

ROOT = Path(__file__).resolve().parent
CAP = 188743680
class BudgetAcquirer(Acquirer):
    def __init__(self, store):
        self.ledger_path = ROOT / 'http-budget.json'
        self.ledger = json.loads(self.ledger_path.read_text()) if self.ledger_path.exists() else {'cap_bytes':CAP,'newly_decoded_bytes':0,'events':[]}
        super().__init__(store,max_bytes=CAP-self.ledger['newly_decoded_bytes']-1048576,delay=1.1)
        self.depth=0
    @staticmethod
    def official(url):
        host=urlsplit(url).hostname or ''
        if not (host=='njleg.gov' or host.endswith('.njleg.gov') or host=='nj.gov' or host.endswith('.nj.gov') or host=='state.nj.us' or host.endswith('.state.nj.us')):
            raise AcquisitionError('Out-of-scope host: '+host)
    def _allowed(self,url):
        self.official(url)
        return super()._allowed(url)
    def fetch(self,url,**kwargs):
        self.official(url)
        if shutil.disk_usage(ROOT).free < 100*1024**3:
            raise AcquisitionError('Free disk below 100 GiB threshold')
        if self.max_bytes-self.downloaded < 1048576:
            raise AcquisitionError('Budget reserve reached')
        start=self.downloaded
        outer=self.depth==0
        self.depth+=1
        try:
            kwargs['max_file_bytes']=min(kwargs.get('max_file_bytes',32*1024**2),self.max_bytes-self.downloaded)
            return super().fetch(url,**kwargs)
        finally:
            self.depth-=1
            if outer:
                delta=self.downloaded-start
                self.ledger['newly_decoded_bytes']+=delta
                self.ledger['events'].append({'at':utc_now(),'url':url,'newly_decoded_bytes':delta})
                tmp=self.ledger_path.with_suffix('.tmp')
                tmp.write_text(json.dumps(self.ledger,indent=2))
                tmp.replace(self.ledger_path)

# The actual sync uses publisher RTF headnote styles to locate boundaries in the
# companion plain text. No section-body reflow, date inference or OCR is used.
import re
import zipfile
import hashlib
from collections import Counter
from psephos.store import Provision

BASE='https://pub.njleg.gov/statutes/'
INVENTORY='https://www.njleg.state.nj.us/api/downloads/|statutes/pub.njleg.gov/Statutes'
DISCLAIMER='https://www.njleg.state.nj.us/disclaimer'

def rtf_headnote_text(raw):
    raw=raw.replace('\r','').replace('\n','')
    raw=re.sub(r"\\'([a-fA-F0-9]{2})",lambda m:bytes.fromhex(m[1]).decode('cp1252'),raw)
    raw=raw.replace('\\line ','\r\n').replace('\\tab ','\t')
    raw=re.sub(r'\\[a-zA-Z]+-?\d* ?', '', raw)
    return raw.replace('{','').replace('}','').strip()

def statute_structure(txt,rtf):
    """Use explicit RTF s3 (Headnotes); fail if any nonblank headnote cannot map."""
    heads=[];cursor=0;blank=0
    for raw in re.findall(r'\\s3 (.*?)\\par',rtf,re.S):
        heading=rtf_headnote_text(raw)
        if not heading:
            blank+=1;continue
        at=txt.find(heading,cursor)
        if at<0:raise ValueError('RTF headnote absent from companion TXT: '+repr(heading))
        heads.append((at,heading));cursor=at+len(heading)
    if len(heads)<10000:raise ValueError('Statute export lacks expected substantive headnotes')
    titles=list(re.finditer(r'^(?:TITLE\s+(\d+[A-Z]?)\s+([^\r\n]+)|APPENDIX ([A-Z])\s+([^\r\n]+))',txt,re.M))
    volumes=[{'key':'preface','title':'Statute export preface','start':0,'end':titles[0].start(),'kind':'publisher_preface'}]
    for i,m in enumerate(titles):
        key=m[1] or 'App.'+m[3]
        volumes.append({'key':key,'title':m[0].strip(),'start':m.start(),'end':titles[i+1].start() if i+1<len(titles) else len(txt),'kind':'statutory_title' if m[1] else 'statutory_appendix'})
    return volumes,heads,blank

def statute_units(txt,volume,heads):
    doc='nj-statutes:'+volume['key']
    relevant=[(at,h) for at,h in heads if volume['start']<=at<volume['end']]
    points=[(volume['start'],volume['title'],None)]
    for at,h in relevant:
        token=h.split()[0].rstrip('.,')
        if re.match(r'^52:9H [0-9]+\b',h):token=' '.join(h.split()[:2])
        points.append((at,h,token))
    counts=Counter(x[2] for x in points if x[2]);seen=Counter()
    for i,(start,heading,token) in enumerate(points):
        end=points[i+1][0] if i+1<len(points) else volume['end']
        if start==end:continue
        text=txt[start:end]
        if not text.strip():raise ValueError('Unexpected whitespace-only source span')
        kind='section' if token and re.fullmatch(r'(?:\d+[A-Z]?|App\.A):[\w.():-]+',token) else 'source_block' if token else 'scope_notes'
        if token and text.strip()==heading.strip():
            kind='disposition_note' if re.search(r'reallocat|relocat|repeal',text,re.I) else 'section_heading'
        seen[token]+=1
        key='nj-statutes:'+token if token else doc+'/_intro'
        if token and counts[token]>1:key+='/occurrence-'+str(seen[token])
        yield Provision(key=key,citation='N.J. Stat. § '+token if token else volume['title'],heading=heading,text=text,markup='',url=BASE+'STATUTES-TEXT.zip',parent_key=doc,unit_kind=kind,metadata={'source_member':'STATUTES.TXT','source_start_byte':start,'source_end_byte':end,'encoding':'windows-1252','boundary_basis':'publisher RTF Headnotes style s3, exact sequential mapping to TXT','source_identifier':token,'duplicate_identifier_count':counts[token] if token else 0,'source_rtf_member':'STATUTES.RTF','warning':'Plain-text projection; companion original RTF retained. Source identifier typos are not silently corrected. Legal effectiveness is not inferred.'})

def sync_nj(s: Store,a: Acquirer,limit=None,as_of=None):
    if as_of:raise ValueError('NJ bulk sync has no historical endpoint or exact incorporation date; as_of is unsupported')
    if a.delay<1:a.delay=1.0
    # The standalone runner supplies BudgetAcquirer; generic callers must supply
    # their own remaining cumulative budget, never a fresh unbounded Acquirer.
    if a.max_bytes>CAP:raise ValueError('Acquirer max_bytes must fit NJ cumulative cap')
    inv=a.fetch(INVENTORY);listing=json.loads(s.artifact(inv.sha256));notice=a.fetch(DISCLAIMER);readme=a.fetch(BASE+'_READ_ME.txt')
    s.collection('nj-statutes',('us-nj','New Jersey','state','us'),name='New Jersey General and Permanent Statutes',authority='New Jersey Legislature, Office of Legislative Services',kind='statutes',homepage='https://www.njleg.state.nj.us/downloads',source_status='Official state publisher bulk text; unannotated; exact incorporation date unknown',access='Public download; robots respected; publisher disclaimer and read-me retained; no blanket redistribution license inferred',metadata={'inventory_artifact':inv.sha256,'disclaimer_artifact':notice.sha256,'readme_artifact':readme.sha256,'warning':'ZIP modification dates do not establish legal currency or effective dates. Catalog/LCTOC is not substantive statute text.'})
    raw=a.fetch(BASE+'STATUTES-TEXT.zip',max_file_bytes=90*1024**2)
    z=zipfile.ZipFile(s.object_path(raw.sha256));txtbytes=z.read('STATUTES.TXT');rtfbytes=z.read('STATUTES.RTF');txt=txtbytes.decode('cp1252');rtf=rtfbytes.decode('cp1252')
    volumes,heads,blank=statute_structure(txt,rtf)
    currency=txt.splitlines()[1]
    manifest={'statutes_currency':currency,'statutes_txt_sha256':hashlib.sha256(txtbytes).hexdigest(),'statutes_rtf_sha256':hashlib.sha256(rtfbytes).hexdigest(),'txt_bytes':len(txtbytes),'rtf_bytes':len(rtfbytes),'rtf_headnotes':len(heads),'blank_rtf_headnotes':blank,'volumes':volumes,'publisher_inventory':listing,'no_image_objects_in_rtf':rtf.count('\\pict')==0,'rtf_table_objects':rtf.count('\\trowd')}
    for v in volumes:
        existing=s.db.execute('SELECT 1 FROM documents WHERE id=?',('nj-statutes:'+v['key'],)).fetchone()
        if not existing:s.inventory('nj-statutes',v['key'],BASE+'STATUTES-TEXT.zip','pending')
    for v in volumes[:limit]:
        vid,count,new=s.ingest(collection='nj-statutes',document='nj-statutes:'+v['key'],title=v['title'],url=BASE+'STATUTES-TEXT.zip',acquisition=raw.id,member='STATUTES.TXT',snapshot_date=None,snapshot_basis='Unknown exact publisher incorporation date; verbatim session-level currency in metadata, not inferred from ZIP/HTTP timestamp',parser='nj-rtf-headnotes-1',provisions=statute_units(txt,v,heads),metadata={'currency_notice':currency,'source_start_byte':v['start'],'source_end_byte':v['end'],'source_txt_sha256':manifest['statutes_txt_sha256'],'source_rtf_sha256':manifest['statutes_rtf_sha256'],'source_volume_kind':v['kind'],'inventory_artifact':inv.sha256,'publisher_file_modified':next(x['DateModified'] for x in listing if x['FileName']=='STATUTES-TEXT.zip')})
        s.inventory('nj-statutes',v['key'],BASE+'STATUTES-TEXT.zip','indexed')
        print(json.dumps({'volume':v['key'],'provisions':count,'new':new}),flush=True)
    con=a.fetch(BASE+'NJCONST-TEXT.zip');cz=zipfile.ZipFile(s.object_path(con.sha256));ct=cz.read('NJCONST.TXT').decode('cp1252')
    articles=list(re.finditer(r'^[\t ]*ARTICLE ([IVX]+)[\t ]*\r?$',ct,re.M))
    if len(articles)!=11:raise ValueError('Constitution expected 11 source articles')
    units=[];starts=[(0,'preamble')]+[(m.start(),m[1]) for m in articles]
    for i,(start,name) in enumerate(starts):
        end=starts[i+1][0] if i+1<len(starts) else len(ct)
        units.append(Provision(key='nj-constitution:'+name,citation='N.J. Const. '+('preamble' if name=='preamble' else 'art. '+name),heading='New Jersey Constitution '+name,text=ct[start:end],markup='',url=BASE+'NJCONST-TEXT.zip',unit_kind='preamble' if name=='preamble' else 'article',metadata={'source_member':'NJCONST.TXT','source_start_byte':start,'source_end_byte':end,'encoding':'windows-1252','warning':'Article-level units retain all sections, paragraphs, amendment/effective-date annotations; companion RTF preserved.'}))
    s.collection('nj-constitution',('us-nj','New Jersey','state','us'),name='New Jersey Constitution 1947',authority='New Jersey Legislature, Office of Legislative Services',kind='constitution',homepage='https://www.njleg.state.nj.us/constitution',source_status='Official state publisher bulk text',access='Public publisher download; disclaimer retained',metadata={'currency_notice':ct.splitlines()[1],'currency_month':'2021-11','disclaimer_artifact':notice.sha256})
    s.ingest(collection='nj-constitution',document='nj-constitution:1947',title='New Jersey Constitution 1947',url=BASE+'NJCONST-TEXT.zip',acquisition=con.id,member='NJCONST.TXT',snapshot_date=None,snapshot_basis='Publisher says updated through amendments adopted November 2021; day unknown, adoption not blanket effectiveness',parser='nj-constitution-articles-1',provisions=units,metadata={'currency_notice':ct.splitlines()[1],'currency_month':'2021-11','publisher_file_modified':next(x['DateModified'] for x in listing if x['FileName']=='NJCONST-TEXT.zip')})
    s.inventory('nj-constitution','1947',BASE+'NJCONST-TEXT.zip','indexed')
    toc=a.fetch(BASE+'LCTOC-TEXT.zip');tz=zipfile.ZipFile(s.object_path(toc.sha256));tt=tz.read('LCTOC.TXT').decode('cp1252')
    manifest['lctoc_currency']=next(x for x in tt.splitlines() if x.strip());manifest['lctoc_counted_as_full_text']=False
    (ROOT/'manifest.json').write_text(json.dumps(manifest,indent=2))
    write_report(s)
